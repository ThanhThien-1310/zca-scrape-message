# `bot:chat` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Một tiến trình `bot:chat` đọc tin nhắn chữ mới trong các nhóm được liệt kê trong `chat-rules.json` và trả lời bằng câu có sẵn theo tình huống (ví dụ báo cáo có máy lỗi), qua giao diện Zalo Desktop.

**Architecture:** `bot:watch` vẫn là nơi duy nhất nghe tin (WebSocket) và chia mỗi tin thành hai luồng: hàng chờ cào (không đổi) và hộp thư chat `data/.chat-inbox.jsonl`. `bot:chat` đọc hộp thư, khớp tình huống (hàm thuần), qua giới hạn tốc độ, rồi gửi bằng CDP với chốt chặn kép trước mỗi lần gõ. Tin do bot gửi ra được ghi vào sổ để `bot:watch` bỏ qua khi nó dội lại. Khóa `data/.scrape.lock` dùng chung với `bot:runner`.

**Tech Stack:** Node ≥22, TypeScript chạy bằng `tsx`, `node:test`, ESM (`.js` khi import), thụt lề 4 dấu cách. Không thêm thư viện.

**Spec:** `docs/superpowers/specs/2026-09-25-bot-chat-design.md` (mẫu kịch bản: `docs/superpowers/specs/2026-09-25-bot-chat-rules.sample.json`)

```ts
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import type { LoaiBaoCao } from "../lesson/types.js";

/** Điều kiện của một tình huống: mọi khóa có mặt phải cùng khớp. */
export interface When {
    loaiBaoCao?: LoaiBaoCao[];
    mayLoi?: true;
    keywords?: string[];
}

export interface Rule {
    id: string;
    when: When;
    reply: string;
}

export interface Limits {
    maxRepliesPerMinute: number;
    maxReplyDelaySeconds: number;
    cooldownSecondsPerRule: number;
}

export interface ChatRules {
    version: 1;
    dryRun: boolean;
    limits: Limits;
    groups: Record<string, { rules: Rule[] }>;
}

export const DEFAULT_LIMITS: Limits = { maxRepliesPerMinute: 6, maxReplyDelaySeconds: 180, cooldownSecondsPerRule: 60 };

const repoRoot = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
export const DEFAULT_RULES_PATH = path.join(repoRoot, "chat-rules.json");

const WHEN_KEYS = ["loaiBaoCao", "mayLoi", "keywords"];
const isObj = (v: unknown): v is Record<string, unknown> => typeof v === "object" && v !== null && !Array.isArray(v);

function parseRule(r: unknown, where: string, seen: Set<string>, errors: string[]): Rule | null {
    if (!isObj(r)) {
        errors.push(`${where}: phải là một đối tượng.`);
        return null;
    }
    const before = errors.length;

    if (typeof r.id !== "string" || r.id.trim() === "") errors.push(`${where}: thiếu "id".`);
    else if (seen.has(r.id)) errors.push(`${where}: "id" "${r.id}" bị trùng trong nhóm.`);
    else seen.add(r.id);

    const when: When = {};
    if (!isObj(r.when) || Object.keys(r.when).length === 0) {
        errors.push(`${where}: "when" phải có ít nhất một điều kiện.`);
    } else {
        const w = r.when;
        for (const k of Object.keys(w)) if (!WHEN_KEYS.includes(k)) errors.push(`${where}: "when" có khóa lạ "${k}".`);
        if ("loaiBaoCao" in w) {
            const v = w.loaiBaoCao;
            if (!Array.isArray(v) || v.length === 0 || !v.every((x) => x === "DauBuoi" || x === "CuoiBuoi")) errors.push(`${where}: "loaiBaoCao" phải là mảng gồm "DauBuoi" và/hoặc "CuoiBuoi".`);
            else when.loaiBaoCao = v as LoaiBaoCao[];
        }
        if ("mayLoi" in w) {
            if (w.mayLoi !== true) errors.push(`${where}: "mayLoi" chỉ nhận true.`);
            else when.mayLoi = true;
        }
        if ("keywords" in w) {
            const v = w.keywords;
            if (!Array.isArray(v) || v.length === 0 || !v.every((x) => typeof x === "string" && x.trim() !== "")) errors.push(`${where}: "keywords" phải là mảng chuỗi không rỗng.`);
            else when.keywords = v as string[];
        }
    }

    if (typeof r.reply !== "string" || r.reply.trim() === "") errors.push(`${where}: thiếu "reply".`);
    else for (const m of r.reply.matchAll(/\{([^}]*)\}/g)) if (m[1] !== "senderName") errors.push(`${where}: "reply" có biến lạ {${m[1]}} (chỉ có {senderName}).`);

    return errors.length > before ? null : { id: r.id as string, when, reply: r.reply as string };
}

/** Kiểm chặt: mọi lỗi được gom lại và ném một lần. `targets` là `config.targets`. */
export function parseRules(json: unknown, targets: string[]): ChatRules {
    if (!isObj(json)) throw new Error("chat-rules.json phải là một đối tượng JSON.");
    const errors: string[] = [];

    if (json.version !== 1) errors.push(`"version" phải là 1 (đang là ${JSON.stringify(json.version)}).`);

    const dryRun = json.dryRun === undefined ? false : json.dryRun;
    if (typeof dryRun !== "boolean") errors.push('"dryRun" phải là true hoặc false.');

    const limits: Limits = { ...DEFAULT_LIMITS };
    if (json.limits !== undefined) {
        if (!isObj(json.limits)) {
            errors.push('"limits" phải là một đối tượng.');
        } else {
            for (const k of Object.keys(json.limits)) if (!(k in DEFAULT_LIMITS)) errors.push(`limits có khóa lạ "${k}".`);
            for (const k of Object.keys(DEFAULT_LIMITS) as Array<keyof Limits>) {
                const v = json.limits[k];
                if (v === undefined) continue;
                if (typeof v !== "number" || !Number.isFinite(v) || v <= 0) errors.push(`limits.${k} phải là số dương.`);
                else limits[k] = v;
            }
        }
    }

    const groups: ChatRules["groups"] = {};
    if (!isObj(json.groups)) {
        errors.push('"groups" phải là một đối tượng (tên nhóm -> { rules }).');
    } else {
        for (const [name, g] of Object.entries(json.groups)) {
            if (!targets.includes(name)) {
                errors.push(`nhóm "${name}" không có trong config.targets.`);
                continue;
            }
            if (!isObj(g) || !Array.isArray(g.rules) || g.rules.length === 0) {
                errors.push(`nhóm "${name}" cần "rules" là mảng không rỗng.`);
                continue;
            }
            const seen = new Set<string>();
            const rules: Rule[] = [];
            g.rules.forEach((r, i) => {
                const rule = parseRule(r, `nhóm "${name}", quy tắc #${i + 1}`, seen, errors);
                if (rule) rules.push(rule);
            });
            groups[name] = { rules };
        }
    }

    if (errors.length > 0) throw new Error(`chat-rules.json không hợp lệ:\n- ${errors.join("\n- ")}`);
    return { version: 1, dryRun: dryRun as boolean, limits, groups };
}

export function loadRules(file: string = DEFAULT_RULES_PATH, targets: string[]): ChatRules {
    let text: string;
    try {
        text = fs.readFileSync(file, "utf8");
    } catch (err) {
        throw new Error(`không đọc được ${file}: ${(err as Error).message}`);
    }
    let json: unknown;
    try {
        json = JSON.parse(text);
    } catch (err) {
        throw new Error(`${file} không phải JSON hợp lệ: ${(err as Error).message}`);
    }
    return parseRules(json, targets);
}

/** Đọc dễ dãi cho `bot:watch`: không bao giờ ném lỗi; tệp thiếu hoặc hỏng thì coi như không có nhóm nào. */
export function groupNamesFromFile(file: string = DEFAULT_RULES_PATH): string[] {
    try {
        const v: unknown = JSON.parse(fs.readFileSync(file, "utf8"));
        return isObj(v) && isObj(v.groups) ? Object.keys(v.groups) : [];
    } catch {
        return [];
    }
}

/** Trả về hàm đọc tập tên nhóm, chỉ đọc lại tệp khi thời điểm sửa đổi thay đổi. */
export function makeGroupNameReader(file: string = DEFAULT_RULES_PATH): () => ReadonlySet<string> {
    let mtime = -1;
    let cached: Set<string> = new Set();
    return () => {
        let m: number;
        try {
            m = fs.statSync(file).mtimeMs;
        } catch {
            m = -2;
        }
        if (m !== mtime) {
            mtime = m;
            cached = new Set(m === -2 ? [] : groupNamesFromFile(file));
        }
        return cached;
    };
}
```

import { classifyTitle, stripDiacritics } from "../lesson/parseReport.js";
import type { InboxEntry } from "./inbox.js";
import type { Rule, When } from "./rules.js";

/** Bỏ dấu, hạ chữ thường, gộp khoảng trắng: dùng để so từ khóa. */
export const foldText = (s: string): string => stripDiacritics(s).toLowerCase().replace(/\s+/g, " ").trim();

/**
 * Báo cáo có máy lỗi: số máy lỗi > 0, hoặc dòng đó có chữ nhưng không phải số (ví dụ "IT bảo trì").
 * "Tổng số máy bị lỗi: 0" và dòng bị thiếu thì không.
 */
export function hasMayLoi(raw: RawFields): boolean {
    if (raw.loi !== null) return raw.loi > 0;
    return raw.loiText !== null && raw.loiText.trim() !== "";
}

/** Mọi khóa có mặt trong `when` phải cùng khớp. */
export function matches(when: When, entry: InboxEntry): boolean {
    if (when.loaiBaoCao) {
        const kind = classifyTitle(entry.text);
        if ((kind !== "DauBuoi" && kind !== "CuoiBuoi") || !when.loaiBaoCao.includes(kind)) return false;
    }
    if (when.mayLoi && !hasMayLoi(extractRaw(entry.text, entry.groupName))) return false;
    if (when.keywords) {
        const text = foldText(entry.text);
        if (!when.keywords.some((k) => text.includes(foldText(k)))) return false;
    }
    return true;
}

/** Tình huống đầu tiên khớp (theo thứ tự khai báo), hoặc null. */
export function pickRule(rules: Rule[], entry: InboxEntry): Rule | null {
    return rules.find((r) => matches(r.when, entry)) ?? null;
}

export function renderReply(rule: Rule, entry: InboxEntry): string {
    return rule.reply.replace(/\{senderName\}/g, entry.senderName.trim() || "bạn");
}

import type { Limits } from "./rules.js";

export type LimitCheck = { ok: true } | { ok: false; reason: "cooldown" | "rate" };

export interface Limiter {
    check(group: string, ruleId: string, now: number): LimitCheck;
    /** Ghi nhận một câu trả lời (thật hoặc dry-run). */
    record(group: string, ruleId: string, now: number): void;
}

/** Trong bộ nhớ: một tình huống không lặp lại trong cùng nhóm quá nhanh, và tổng số tin mỗi phút bị chặn. */
export function createLimiter(limits: Limits): Limiter {
    const last = new Map<string, number>();
    let sent: number[] = [];
    return {
        check(group, ruleId, now) {
            const at = last.get(`${group}|${ruleId}`);
            if (at !== undefined && now - at < limits.cooldownSecondsPerRule * 1000) return { ok: false, reason: "cooldown" };
            sent = sent.filter((t) => now - t < 60_000);
            if (sent.length >= limits.maxRepliesPerMinute) return { ok: false, reason: "rate" };
            return { ok: true };
        },
        record(group, ruleId, now) {
            last.set(`${group}|${ruleId}`, now);
            sent.push(now);
        },
    };
}

/** Trả lời quá trễ có thể còn tệ hơn không trả lời. */
export function isStale(messageTs: number, now: number, maxDelaySeconds: number): boolean {
    return now - messageTs > maxDelaySeconds * 1000;
}

import fs from "node:fs";
import path from "node:path";
import { config } from "../config.js";

/** Một tin chữ `bot:watch` đưa vào hộp thư chat. */
export interface InboxEntry {
    groupName: string;
    groupId: string;
    msgId: string;
    senderId: string;
    senderName: string;
    ts: number;
    text: string;
}

/** Một dòng đã đọc, kèm vị trí byte ngay sau nó (đặt con trỏ tới đó để coi như đã xử lý). */
export interface InboxItem {
    entry: InboxEntry;
    end: number;
}

/**
 * Hộp thư của `bot:chat`: `bot:watch` chỉ thêm dòng, `bot:chat` đọc từ con trỏ. Tách khỏi hàng chờ cào
 * để `bot:runner` không bao giờ nhìn thấy nó. Mỗi dòng là một JSON; dòng chưa xuống dòng (đang ghi dở) chưa được đọc.
 */
export const DEFAULT_INBOX_PATH = path.join(config.dataDir, ".chat-inbox.jsonl");
export const DEFAULT_CURSOR_PATH = path.join(config.dataDir, ".chat-inbox.cursor");

export function appendInbox(entry: InboxEntry, file: string = DEFAULT_INBOX_PATH): void {
    fs.mkdirSync(path.dirname(file), { recursive: true });
    fs.appendFileSync(file, `${JSON.stringify(entry)}\n`, "utf8");
}

function readCursor(cursorFile: string): number {
    try {
        const n = Number(fs.readFileSync(cursorFile, "utf8").trim());
        return Number.isInteger(n) && n >= 0 ? n : 0;
    } catch {
        return 0;
    }
}

export function writeCursor(offset: number, cursorFile: string = DEFAULT_CURSOR_PATH): void {
    fs.mkdirSync(path.dirname(cursorFile), { recursive: true });
    const tmp = `${cursorFile}.tmp`;
    fs.writeFileSync(tmp, String(offset), "utf8");
    fs.renameSync(tmp, cursorFile);
}

function isEntry(v: unknown): v is InboxEntry {
    if (typeof v !== "object" || v === null) return false;
    const o = v as Record<string, unknown>;
    return ["groupName", "groupId", "msgId", "senderId", "senderName", "text"].every((k) => typeof o[k] === "string") && typeof o.ts === "number";
}

/** Các dòng đầy đủ sau con trỏ, theo thứ tự; dòng hỏng bị bỏ qua. Con trỏ vượt quá tệp (tệp đã xoay) thì đọc lại từ đầu. */
export function readInbox(file: string = DEFAULT_INBOX_PATH, cursorFile: string = DEFAULT_CURSOR_PATH): InboxItem[] {
    let buf: Buffer;
    try {
        buf = fs.readFileSync(file);
    } catch {
        return [];
    }
    let pos = readCursor(cursorFile);
    if (pos > buf.length) pos = 0;

    const items: InboxItem[] = [];
    while (pos < buf.length) {
        const nl = buf.indexOf(0x0a, pos);
        if (nl < 0) break; // dòng cuối chưa hoàn chỉnh
        const line = buf.subarray(pos, nl).toString("utf8");
        pos = nl + 1;
        try {
            const v: unknown = JSON.parse(line);
            if (isEntry(v)) items.push({ entry: v, end: pos });
        } catch {
            /* dòng hỏng */
        }
    }
    return items;
}
```

import path from "node:path";
import { config } from "../config.js";

export const DEFAULT_DONE_PATH = path.join(config.dataDir, ".chat-done.json");
export const DEFAULT_SENT_PATH = path.join(config.dataDir, ".chat-sent.json");

const KEEP_SENT_MS = 10 * 60_000;
const ECHO_WINDOW_MS = 2 * 60_000;

function writeAtomic(file: string, value: unknown): void {
    fs.mkdirSync(path.dirname(file), { recursive: true });
    const tmp = `${file}.tmp`;
    fs.writeFileSync(tmp, JSON.stringify(value, null, 2), "utf8");
    fs.renameSync(tmp, file);
}

function readJson(file: string): unknown {
    try {
        return JSON.parse(fs.readFileSync(file, "utf8"));
    } catch {
        return undefined;
    }
}

/** `msgId` của các tin đã xử lý (đã trả lời, hoặc quyết định không trả lời): chống trả lời lại sau khi khởi động lại. */
export function loadDone(file: string = DEFAULT_DONE_PATH): Set<string> {
    const v = readJson(file);
    return new Set(Array.isArray(v) ? v.filter((x): x is string => typeof x === "string") : []);
}

export function markDone(msgId: string, file: string = DEFAULT_DONE_PATH, keep = 500): void {
    const ids = [...loadDone(file)].filter((id) => id !== msgId);
    ids.push(msgId);
    writeAtomic(file, ids.slice(-keep));
}

interface Sent {
    groupName: string;
    text: string;
    at: number;
}

const norm = (s: string): string => s.normalize("NFC").replace(/\s+/g, " ").trim();

function readSent(file: string): Sent[] {
    const v = readJson(file);
    return Array.isArray(v) ? v.filter((s): s is Sent => !!s && typeof s.groupName === "string" && typeof s.text === "string" && typeof s.at === "number") : [];
}

/** Ghi nhớ một câu sắp gửi (ghi TRƯỚC khi gửi, để tin dội lại không kịp đến trước sổ). Các mục quá 10 phút bị xóa. */
export function recordSent(groupName: string, text: string, at: number, file: string = DEFAULT_SENT_PATH): void {
    const kept = readSent(file).filter((s) => at - s.at < KEEP_SENT_MS);
    kept.push({ groupName, text: norm(text), at });
    writeAtomic(file, kept);
}

/**
 * `bot:watch` hỏi khi thấy một tin của chính tài khoản: có phải câu bot vừa gửi không? Nếu có (cùng nhóm, cùng nội
 * sau khi chuẩn hóa, trong 2 phút), mục đó bị tiêu (mỗi câu đã gửi chỉ khớp một tin dội lại) và trả về true.
 */
export function consumeEcho(groupName: string, text: string, now: number, file: string = DEFAULT_SENT_PATH, windowMs: number = ECHO_WINDOW_MS): boolean {
    const sent = readSent(file);
    const wanted = norm(text);
    const i = sent.findIndex((s) => s.groupName === groupName && s.text === wanted && now - s.at <= windowMs);
    if (i < 0) return false;
    sent.splice(i, 1);
    writeAtomic(file, sent);
    return true;
}

import type { RealtimeMessageEvent } from "../history/realtimeListener.js";
import type { InboxEntry } from "./inbox.js";

export interface RouteInput {
    event: Pick<RealtimeMessageEvent, "groupId" | "photoMessage" | "textMessage">;
    /** Tên nhóm trong `config.targets` ứng với `groupId`; undefined nếu chưa biết. */
    target: string | undefined;
    /** Nhóm có trong `chat-rules.json`. */
    allowedGroups: ReadonlySet<string>;
    /** Có phải câu bot vừa gửi quay về không (tiêu mục trong sổ nếu đúng). */
    consumeEcho: (target: string, text: string) => boolean;
}

export interface Route {
    /** Xếp nhóm vào hàng chờ cào (luồng của `bot:runner`). */
    queue: boolean;
    /** Dòng ghi vào hộp thư chat (luồng của `bot:chat`), hoặc null. */
    inbox: InboxEntry | null;
}

/**
 * Một tin đi vào hai luồng độc lập. Nhóm không có trong rules không bao giờ vào hộp thư chat; tin của chính
 * tài khoản không bao giờ vào hộp thư chat; câu bot vừa gửi quay về bị bỏ ở cả hai luồng (không cào thừa, không tự trả lời).
 */
export function routeMessage(input: RouteInput): Route {
    const { event, target, allowedGroups } = input;
    if (!target) return { queue: false, inbox: null };
    const t = event.textMessage;
    if (t?.isSelf && input.consumeEcho(target, t.content)) return { queue: false, inbox: null };

    const inbox: InboxEntry | null =
        t && !t.isSelf && t.content.trim() !== "" && allowedGroups.has(target)
            ? { groupName: target, groupId: event.groupId, msgId: t.msgId, senderId: t.senderId, senderName: t.senderName, ts: t.ts, text: t.content }
            : null;
    return { queue: true, inbox };
}

 * Chốt chặn kép: tên nhóm đang chọn bằng đúng nhóm đích VÀ placeholder của ô soạn tin kết thúc bằng
 * "tới <nhóm đích>" (đo độc lập với thanh bên). Thiếu một trong hai thì coi như sai nhóm.
 */
export function chatIsTarget(s: ComposerState, target: string): boolean {
    if (s.title === null || s.placeholder === null) return false;
    const want = norm(target);
    const ph = norm(s.placeholder);
    return norm(s.title) === want && (ph === `tới ${want}` || ph.endsWith(` tới ${want}`)) && /(^|\s)tới /.test(ph) && ph.slice(0, ph.length - want.length).trimEnd().endsWith("tới");
}

    const t = document.querySelector(".conv-item.selected .conv-item-title__name");
    const r = document.getElementById("richInput");
    return { title: t ? t.textContent : null, placeholder: r ? r.getAttribute("placeholder") : null, empty: r ? r.classList.contains("empty") : null, text: r ? r.innerText : null };
})()`;

const search = (query: string) => `/*search*/(() => {
    const spec = { "query": ${JSON.stringify(query)} };
    const i = document.querySelector("#contact-search-input");
    if (!i) return false;
    Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, "value").set.call(i, spec.query);
    i.dispatchEvent(new Event("input", { bubbles: true }));
    i.blur();
    return true;
})()`;

const click = (name: string) => `/*click*/(() => {
    const n = (s) => s.normalize("NFC").replace(/[\\u200B-\\u200D\\uFEFF]/g, "").replace(/\\s+/g, " ").trim();
    const w = n(${JSON.stringify(name)});
    const t = Array.from(document.querySelectorAll(".conv-item-title__name")).find((x) => n(x.textContent || "") === w);
    if (!t) return false;
    const it = t.closest(".conv-item");
    if (!it) return false;
    it.click();
    return true;
})()`;

const FOCUS = `/*focus*/(() => { const r = document.getElementById("richInput"); if (!r) return false; r.focus(); r.click(); return true; })()`;

const LAST_ROW = `/*lastRow*/(() => { const rows = document.querySelectorAll(".chat-message"); const last = rows[rows.length - 1]; return last ? (last.innerText || "") : ""; })()`;

/** Chỉ điều hướng (tìm rồi bấm một mục ở thanh bên, xóa ô tìm); không gõ vào cuộc trò chuyện nào. */
async function open(page: SendPage, name: string, deps: SendDeps): Promise<void> {
    try {
        await page.evaluate<boolean>(search(name));
        await deps.delay(1000);
        await page.evaluate<boolean>(click(name));
    } finally {
        await page.evaluate<boolean>(search(""));
    }
}

const key = { key: "Enter", code: "Enter", windowsVirtualKeyCode: 13, nativeVirtualKeyCode: 13 };

/**
 * Gửi một tin vào nhóm `target` qua giao diện Zalo Desktop. Người gọi phải giữ khóa `data/.scrape.lock`.
 * Trước MỖI thao tác gõ hoặc Enter đều đọc lại trạng thái và kiểm `chatIsTarget`; sai thì dừng, không gõ tiếp.
 * Lỗi sau khi đã bấm Enter mà không xác nhận được thì báo `unconfirmed` và KHÔNG tự gửi lại (tránh gửi trùng).
 * Luôn mở lại cuộc trò chuyện đang mở trước đó (chỉ điều hướng).
 */
export async function sendReply(page: SendPage, target: string, text: string, deps: SendDeps): Promise<SendResult> {
    const read = () => page.evaluate<ComposerState>(STATE);
    const guarded = async (): Promise<ComposerState | null> => {
        const s = await read();
        return chatIsTarget(s, target) ? s : null;
    };

    const previous = (await read()).title;
    try {
        await open(page, target, deps);
        await deps.delay(1200);

        const opened = await guarded();
        if (!opened) return { status: "skipped", reason: "not-opened" };
        if (opened.empty !== true) return { status: "skipped", reason: "draft" };

        await page.evaluate<boolean>(FOCUS);
        await deps.delay(300);

        if (!(await guarded())) return { status: "failed", reason: "nhóm đang mở đã đổi trước khi gõ" };
        await page.input("Input.insertText", { text });
        await deps.delay(500);

        let after = await read();
        if (!(after.text ?? "").includes(text.split("\n")[0])) {
            deps.log("[chat] insertText không vào ô soạn tin, thử gửi từng ký tự");
            if (!(await guarded())) return { status: "failed", reason: "nhóm đang mở đã đổi trước khi gõ từng ký tự" };
            for (const ch of text) await page.input("Input.dispatchKeyEvent", ch === "\n" ? { type: "char", text: "\n", key: "Enter", modifiers: 8 } : { type: "char", text: ch });
            await deps.delay(500);
            after = await read();
        }
        if (!chatIsTarget(after, target)) return { status: "failed", reason: "nhóm đang mở đã đổi sau khi gõ; không gửi" };
        if (!(after.text ?? "").includes(text.split("\n")[0])) return { status: "failed", reason: "không nhập được chữ vào ô soạn tin; chưa gửi gì" };

        if (!(await guarded())) return { status: "failed", reason: "nhóm đang mở đã đổi trước khi bấm Enter" };
        await page.input("Input.dispatchKeyEvent", { type: "rawKeyDown", ...key });
        await page.input("Input.dispatchKeyEvent", { type: "char", text: "\r", ...key });
        await page.input("Input.dispatchKeyEvent", { type: "keyUp", ...key });
        await deps.delay(1500);

        const final = await read();
        const lastRow = await page.evaluate<string>(LAST_ROW);
        if (final.empty === true && norm(lastRow).includes(norm(text))) return { status: "sent" };
        return { status: "failed", reason: "đã bấm Enter nhưng không xác nhận được tin đã gửi", unconfirmed: true };
    } finally {
        if (previous && norm(previous) !== norm(target)) {
            try {
                await open(page, previous, deps);
            } catch {
                /* điều hướng lại chỉ là tiện lợi */
            }
        }
    }
}
```

```ts
import type { InboxItem } from "./inbox.js";
import { isStale, type Limiter } from "./limits.js";
import { pickRule, renderReply } from "./match.js";
import type { ChatRules } from "./rules.js";
import type { SendResult } from "./send.js";

export interface ChatDeps {
    now(): number;
    rules: ChatRules;
    limiter: Limiter;
    /** Các dòng hộp thư chưa xử lý, theo thứ tự. */
    readNew(): InboxItem[];
    /** Đặt con trỏ tới vị trí byte `end` (mọi dòng trước đó coi như xong). */
    commit(end: number): void;
    isDone(msgId: string): boolean;
    markDone(msgId: string): void;
    /** Ghi câu sắp gửi vào sổ tin dội lại; gọi TRƯỚC khi gửi. */
    recordSent(group: string, text: string, at: number): void;
    /** Chạy `fn` khi giữ khóa giao diện Zalo; `acquired: false` khi người khác đang giữ. */
    withLock(fn: () => Promise<SendResult>): Promise<{ acquired: true; result: SendResult } | { acquired: false }>;
    send(group: string, text: string): Promise<SendResult>;
    /** Công tắc dừng nhanh (tệp `data/.chat-off`). */
    off(): boolean;
    log(message: string): void;
}

export interface ChatSummary {
    sent: number;
    dryRun: number;
    /** Không trả lời (không khớp, nhóm không có rules, quá hạn, cooldown, công tắc dừng): coi như xong. */
    ignored: number;
    /** Chưa gửi được lúc này (khóa bận, ô soạn tin đang có chữ, quá tốc độ): giữ lại cho vòng sau. */
    deferred: number;
    failed: number;
}

/**
 * Xử lý các tin mới của hộp thư theo thứ tự. Dừng ở tin đầu tiên phải hoãn để giữ đúng thứ tự;
 * con trỏ chỉ tiến tới hết tin cuối đã xử lý xong. Tin gửi lỗi (kể cả không xác nhận được) được đánh dấu xong và
 * không tự gửi lại, vì tin có thể đã ra ngoài.
 */
export async function runChatCycle(deps: ChatDeps): Promise<ChatSummary> {
    const summary: ChatSummary = { sent: 0, dryRun: 0, ignored: 0, deferred: 0, failed: 0 };
    let advanceTo: number | null = null;

    for (const { entry, end } of deps.readNew()) {
        const now = deps.now();

        if (deps.isDone(entry.msgId)) {
            advanceTo = end;
            continue;
        }
        const group = deps.rules.groups[entry.groupName];
        if (!group) {
            summary.ignored++;
            advanceTo = end;
            continue;
        }
        if (isStale(entry.ts, now, deps.rules.limits.maxReplyDelaySeconds)) {
            deps.log(`[chat] bỏ tin ${entry.msgId} của "${entry.groupName}": quá hạn trả lời`);
            summary.ignored++;
            advanceTo = end;
            continue;
        }
        const rule = pickRule(group.rules, entry);
        if (!rule) {
            summary.ignored++;
            advanceTo = end;
            continue;
        }
        if (deps.off()) {
            deps.log(`[chat] công tắc dừng đang bật: không trả lời tin ${entry.msgId} ("${rule.id}")`);
            summary.ignored++;
            advanceTo = end;
            continue;
        }

        const check = deps.limiter.check(entry.groupName, rule.id, now);
        if (!check.ok && check.reason === "cooldown") {
            deps.log(`[chat] "${rule.id}" ở "${entry.groupName}" đang trong thời gian nghỉ: bỏ tin ${entry.msgId}`);
            summary.ignored++;
            advanceTo = end;
            continue;
        }
        if (!check.ok) {
            deps.log("[chat] đã chạm giới hạn số tin mỗi phút: hoãn phần còn lại");
            summary.deferred++;
            break;
        }

        const text = renderReply(rule, entry);
        if (deps.rules.dryRun) {
            deps.log(`[chat] [dry-run] sẽ trả lời "${entry.groupName}" (${rule.id}): ${text}`);
            deps.limiter.record(entry.groupName, rule.id, now);
            deps.markDone(entry.msgId);
            summary.dryRun++;
            advanceTo = end;
            continue;
        }

        deps.recordSent(entry.groupName, text, now);
        let outcome: { acquired: true; result: SendResult } | { acquired: false };
        try {
            outcome = await deps.withLock(() => deps.send(entry.groupName, text));
        } catch (err) {
            outcome = { acquired: true, result: { status: "failed", reason: (err as Error).message } };
        }
        if (!outcome.acquired) {
            deps.log("[chat] Zalo đang được tiến trình khác điều khiển: hoãn");
            summary.deferred++;
            break;
        }
        const result = outcome.result;
        if (result.status === "sent") {
            deps.log(`[chat] đã trả lời "${entry.groupName}" (${rule.id}): ${text}`);
            deps.limiter.record(entry.groupName, rule.id, now);
            deps.markDone(entry.msgId);
            summary.sent++;
            advanceTo = end;
        } else if (result.status === "skipped") {
            deps.log(`[chat] chưa gửi được vào "${entry.groupName}" (${result.reason}): hoãn`);
            summary.deferred++;
            break;
        } else {
            deps.log(`[chat] gửi lỗi vào "${entry.groupName}": ${result.reason}${result.unconfirmed ? " (có thể đã gửi; không gửi lại)" : ""}`);
            deps.markDone(entry.msgId);
            summary.failed++;
            advanceTo = end;
        }
    }

    if (advanceTo !== null) deps.commit(advanceTo);
    return summary;
}
```

```ts
import fs from "node:fs";
import path from "node:path";
import { runChatCycle } from "./chat/cycle.js";
import { DEFAULT_CURSOR_PATH, DEFAULT_INBOX_PATH, readInbox, writeCursor } from "./chat/inbox.js";
import { createLimiter } from "./chat/limits.js";
import { DEFAULT_RULES_PATH, loadRules, type ChatRules } from "./chat/rules.js";
import { sendReply, toSendPage } from "./chat/send.js";
import { loadDone, markDone, recordSent } from "./chat/state.js";
import { config } from "./config.js";
import { connectToZaloMainPage } from "./history/cdp.js";
import { acquireScrapeLock, releaseScrapeLock } from "./history/scrapeLock.js";

/**
 * Trả lời tự động theo kịch bản `chat-rules.json`. Chỉ ĐỌC hộp thư `data/.chat-inbox.jsonl` (do `bot:watch` ghi)
 * và GỬI qua giao diện Zalo Desktop; không nghe WebSocket, không cào. Mỗi lần gửi giữ khóa `data/.scrape.lock`
 * chung với `bot:runner`. `dryRun: true` trong tệp kịch bản chỉ ghi log. Tạo tệp `data/.chat-off` để dừng gửi ngay.
 *
 *   npm run bot:chat
 */
const POLL_MS = 3_000;
const HEARTBEAT_MS = 30_000;
const STATE_PATH = path.join(config.dataDir, ".chat-state.json");
const OFF_PATH = path.join(config.dataDir, ".chat-off");
const delay = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));
const log = (m: string) => console.log(`${new Date().toISOString()} ${m}`);

function writeState(value: object): void {
    fs.mkdirSync(path.dirname(STATE_PATH), { recursive: true });
    const tmp = `${STATE_PATH}.tmp`;
    fs.writeFileSync(tmp, JSON.stringify(value, null, 2), "utf8");
    fs.renameSync(tmp, STATE_PATH);
}

function rulesMtime(): number {
    try {
        return fs.statSync(DEFAULT_RULES_PATH).mtimeMs;
    } catch {
        return -1;
    }
}

async function main(): Promise<void> {
    let rules: ChatRules;
    try {
        rules = loadRules(DEFAULT_RULES_PATH, config.targets);
    } catch (err) {
        console.error((err as Error).message);
        process.exit(1);
    }
    let mtime = rulesMtime();
    let limiter = createLimiter(rules.limits);
    const startedAt = Date.now();
    let lastSummary: object | undefined;

    const beat = () => writeState({ pid: process.pid, startedAt, lastBeat: Date.now(), dryRun: rules.dryRun, groups: Object.keys(rules.groups), lastSummary });
    beat();
    setInterval(beat, HEARTBEAT_MS);
    const stop = () => {
        writeState({ pid: process.pid, startedAt, lastBeat: Date.now(), stopped: true, dryRun: rules.dryRun });
        releaseScrapeLock();
        process.exit(0);
    };
    process.on("SIGINT", stop);
    process.on("SIGTERM", stop);

    log(`[chat] chạy: ${Object.keys(rules.groups).join(", ") || "(chưa có nhóm nào)"} | dryRun=${rules.dryRun}`);

    for (;;) {
        await delay(POLL_MS);

        const m = rulesMtime();
        if (m !== mtime) {
            mtime = m;
            try {
                rules = loadRules(DEFAULT_RULES_PATH, config.targets);
                limiter = createLimiter(rules.limits);
                log(`[chat] đã nạp lại kịch bản: ${Object.keys(rules.groups).join(", ")} | dryRun=${rules.dryRun}`);
            } catch (err) {
                log(`[chat] kịch bản mới không hợp lệ, giữ bản cũ:\n${(err as Error).message}`);
            }
        }

        const done = loadDone();
        const summary = await runChatCycle({
            now: () => Date.now(),
            rules,
            limiter,
            readNew: () => readInbox(DEFAULT_INBOX_PATH, DEFAULT_CURSOR_PATH),
            commit: (end) => writeCursor(end, DEFAULT_CURSOR_PATH),
            isDone: (id) => done.has(id),
            markDone: (id) => {
                done.add(id);
                markDone(id);
            },
            recordSent: (g, t, at) => recordSent(g, t, at),
            withLock: async (fn) => {
                const lock = acquireScrapeLock("bot:chat");
                if (!lock.ok) return { acquired: false };
                try {
                    return { acquired: true, result: await fn() };
                } finally {
                    releaseScrapeLock();
                }
            },
            send: async (group, text) => {
                const page = await connectToZaloMainPage(config.cdpUrl);
                try {
                    return await sendReply(toSendPage(page), group, text, { delay, log });
                } finally {
                    page.close();
                }
            },
            off: () => fs.existsSync(OFF_PATH),
            log,
        });
        if (summary.sent + summary.dryRun + summary.failed + summary.deferred > 0) {
            lastSummary = { at: Date.now(), ...summary };
            log(`[chat] vòng xong: ${JSON.stringify(summary)}`);
        }
    }
}

main().catch((err) => {
    console.error(err);
    releaseScrapeLock();
    process.exit(1);
});
```

- [ ] **Step 2: Viết `chatStatus.ts`**

```ts
import fs from "node:fs";
import path from "node:path";
import { readInbox, DEFAULT_CURSOR_PATH, DEFAULT_INBOX_PATH } from "./chat/inbox.js";
import { config } from "./config.js";
import { STALE_AFTER_MS } from "./history/watchState.js";

/**
 * `bot:chat` còn sống không, đang chạy thật hay dry-run, nhóm nào, còn bao nhiêu tin chờ, công tắc dừng có bật không.
 * Chỉ đọc. Thoát mã 0 khi đang sống, 2 khi không.
 *
 *   npm run chat:status
 */
const now = Date.now();
let state: { pid?: number; startedAt?: number; lastBeat?: number; stopped?: boolean; dryRun?: boolean; groups?: string[]; lastSummary?: unknown } | null = null;
try {
    state = JSON.parse(fs.readFileSync(path.join(config.dataDir, ".chat-state.json"), "utf8"));
} catch {
    state = null;
}
const hhmm = (ts: number) => new Date(ts + 7 * 3600_000).toISOString().slice(11, 16);

let alive = false;
if (!state || typeof state.lastBeat !== "number") console.log("[chat] chưa có dữ liệu: bot:chat chưa từng chạy.");
else if (state.stopped) console.log(`[chat] ĐÃ DỪNG lúc ${hhmm(state.lastBeat)}`);
else if (now - state.lastBeat > STALE_AFTER_MS) console.log(`[chat] CÓ VẺ ĐÃ CHẾT: nhịp tim cuối lúc ${hhmm(state.lastBeat)} (pid ${state.pid}).`);
else {
    alive = true;
    console.log(`[chat] ĐANG CHẠY (pid ${state.pid}) | ${state.dryRun ? "DRY-RUN (không gửi thật)" : "gửi thật"} | nhóm: ${(state.groups ?? []).join(", ") || "(không có)"}`);
    if (state.lastSummary) console.log(`  vòng gần nhất: ${JSON.stringify(state.lastSummary)}`);
}
console.log(`  tin chờ trong hộp thư: ${readInbox(DEFAULT_INBOX_PATH, DEFAULT_CURSOR_PATH).length}`);
if (fs.existsSync(path.join(config.dataDir, ".chat-off"))) console.log("  CÔNG TẮC DỪNG (data/.chat-off) đang BẬT: bot không gửi gì.");
if (!alive) process.exitCode = 2;
```
